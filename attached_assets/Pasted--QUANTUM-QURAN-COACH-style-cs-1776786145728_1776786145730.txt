function writeDB(data) {
  const tmp = DB_PATH + '.tmp';

}

function advancePhase(plan) {
  if (plan.type !== 'quantum_auto') return plan;
  const list = plan.manual_override ? [...PHASE_MAP, ...OVERRIDE_PHASES] : PHASE_MAP;
  const idx  = list.findIndex(p => p.phase === plan.phase);
  if (idx === -1) return plan;
  const cfg  = list[idx];
  if ((plan.phase_days_elapsed || 0) >= cfg.duration) {
    const next = list[Math.min(idx + 1, list.length - 1)];
    plan.phase               = next.phase;
    plan.current_daily_pages = next.pages;
    plan.phase_start_date    = today();
    plan.phase_days_elapsed  = 0;
  }
  return plan;
}

function buildCustomPlan(goalMonths) {
  const days  = (goalMonths || 24) * 30;
  const pages = +Math.min(5, Math.max(0.25, TOTAL_PAGES / days)).toFixed(2);
  return { type:'custom', start_date:today(), current_daily_pages:pages,
           phase:'custom_steady', phase_start_date:today(), phase_days_elapsed:0,
           manual_override:false, override_target:null, weekly_off_days:[] };
}

function projectedCompletion(pagesMemorized, dailyPages) {
  const rem = TOTAL_PAGES - pagesMemorized;
  if (rem <= 0) return today();
  const d = new Date();
  d.setDate(d.getDate() + Math.ceil(rem / dailyPages));
  return d.toISOString().split('T')[0];
}

function assignReviewSystem(sessions, dailyMinutes, juzNum) {
  // Use ai_core if available
  if (aiCore && aiCore.assignReviewSystem) {
    return aiCore.assignReviewSystem(sessions, dailyMinutes, juzNum);
  }
  // Fallback
  const recent   = sessions.slice(-20);
  const total    = Math.max(1, recent.length);
  const hardRatio = recent.filter(s => s.difficulty === 'hard').length   / total;
  const medRatio  = recent.filter(s => s.difficulty === 'medium').length / total;
  if (hardRatio > 0.6 && dailyMinutes < 30) {
    return { type:'full_stop', dedicated_days_per_week:null,
             message_ar:`أتممت الجزء ${juzNum}! توقف مؤقت للتثبيت.`,
             message_en:`Juz ${juzNum} complete! Pausing to solidify.` };
  }
  if (hardRatio > 0.35 || medRatio > 0.5) {
    const days = dailyMinutes >= 45 ? 2 : 3;
    return { type:'dedicated_days', dedicated_days_per_week:days,
             message_ar:`أتممت الجزء ${juzNum}! ${days} أيام للمراجعة.`,
             message_en:`Juz ${juzNum} complete! ${days} days/week review.` };
  }
  return { type:'daily_mix', dedicated_days_per_week:null,
           message_ar:`أتممت الجزء ${juzNum}! مزيج يومي.`,
           message_en:`Juz ${juzNum} complete! Daily mix.` };
}

function selectAlert(user, trigger) {
  if (aiCore && aiCore.AlertSelector) {
    const sel = new aiCore.AlertSelector(alertsPool);
    return sel.select(user, trigger);
  }
  // Simple fallback selector
  const style   = user.onboarding?.learning_style || 'visual';
  const absent  = user.progress?.consecutive_absences || 0;
  const energy  = user.energy?.score || 75;
  let pool = [...alertsPool];
  if (absent >= 3)  pool = pool.filter(a => a.category === 'intervention' || a.trigger?.startsWith('absence'));
  else if (energy < 40) pool = pool.filter(a => a.category === 'breathing' || a.category === 'psychology');
  else if (energy > 80) pool = pool.filter(a => a.intensity === 'high');
  if (trigger) {
    const exact = pool.filter(a => a.trigger === trigger);
    if (exact.length) return exact[Math.floor(Math.random() * exact.length)];
  }
  if (!pool.length) pool = alertsPool;
  return pool[Math.floor(Math.random() * pool.length)] || { text_ar:'استمر، أنت قادر!', text_en:'Keep going!', id:'fallback' };
}

/* ══════════════════════════════════════════════════════════
   PROCESS-STATE — Real-time Quantum AI Pipeline
   Connects frontend sensors → ai_core → UI decision
══════════════════════════════════════════════════════════ */

function runProcessStatePipeline(sensors, userRecord, adminWeights) {
  if (aiCore && aiCore.processState) {
    // Full ai_core pipeline (Kalman, Bayesian, Local ML, etc.)
    return aiCore.processState(sensors, userRecord, adminWeights, alertsPool);
  }

  // ── Fallback heuristic pipeline ──
  const w = adminWeights || {};
  let energy = userRecord.energy?.score || 75;

  // Sensor deductions
  energy -= Math.min(40, (sensors.hesitationMs   || 0) / 1000 * 8);
  energy -= Math.min(20, (sensors.mouseErratics  || 0) * 2);
  energy -= Math.min(25, (sensors.hiddenSeconds  || 0) / 30 * 5);
  energy -= Math.min(30, (sensors.exitAttempts   || 0) * 10);
  energy += Math.min(20, (sensors.focusSeconds   || 0) / 60 * 4);
  if (sensors.scrolledToBottom) energy += 5;

  energy = Math.max(0, Math.min(100, Math.round(energy)));

  let mode = 'NORMAL_MODE';
  if (energy < 38)                                mode = 'RELOAD_MODE';
  else if (energy > 80)                           mode = 'CHALLENGE_MODE';
  if ((userRecord.energy?.afi || 100) < 22)       mode = 'PATTERN_INTERRUPT';
  if (sensors.trigger === 'exit_intent')           mode = 'PATTERN_INTERRUPT';

  const base = userRecord.plan?.current_daily_pages || 0.25;
  let targetPages = base;
  if (mode === 'RELOAD_MODE')       targetPages = Math.max(0.25, base * 0.5);
  if (mode === 'CHALLENGE_MODE')    targetPages = Math.min(base * 2, 5);

  const friction = Math.min(10, (sensors.hesitationMs || 0) / 1000 * 0.9 + (sensors.mouseErratics || 0) * 0.15);

  return {
    mode,
    energy,
    friction:     +friction.toFixed(2),
    afi:          userRecord.energy?.afi || 100,
    decay_rate:   0.02,
    circadian:    1.0,
    target_pages: +targetPages.toFixed(2),
    absence_risk: 0,
    intervention: null,
    alert:        selectAlert(userRecord, sensors.trigger || 'generic_daily'),
    streak:       userRecord.progress?.current_streak_days || 0,
    diagnostics:  { ml_samples: 0, fallback: true },
  };
}

function requireAuth(req, res, next) {
  const username = (req.headers['x-username'] || '').toLowerCase().trim();
  const token    = req.headers['x-token'] || '';
  if (!username || !token) return res.status(401).json({ error:'Unauthenticated' });
  const db   = readDB();
  const user = db.users[username];
  if (!user || user.session_token !== token) return res.status(401).json({ error:'Invalid session' });
  req.username = username;
  req.user     = user;
  req.db       = db;
  next();
}

function requireAdmin(req, res, next) {
  const pwd = req.headers['x-admin-password'] || req.body?.admin_password || '';
  if (sha256(pwd) !== sha256(ADMIN_PASSWORD)) return res.status(403).json({ error:'Forbidden' });
  next();
}

function updateAbsence(user) {
  const last = user.progress.last_session_date;
  if (!last) return;
  const diff = daysBetween(last, today());
  if (diff > 1) {
    user.progress.consecutive_absences = diff - 1;
    user.progress.total_absences      += diff - 1;
    user.progress.current_streak_days  = 0;
  } else {
    user.progress.consecutive_absences = 0;
  }
}

app.post('/api/auth/register', (req, res) => {
  const { username, password, display_name } = req.body;
  if (!username || !password) return res.status(400).json({ error:'Missing fields' });
  const key = username.toLowerCase().trim();
  if (!/^[a-z0-9_]{3,20}$/.test(key)) return res.status(400).json({ error:'Username: 3-20 chars (a-z 0-9 _)' });
  const db = readDB();
  if (db.users[key]) return res.status(409).json({ error:'Username taken' });
  const token = uid() + uid();
  const user  = blankUser(key, sha256(password), display_name);
  user.session_token = token;
  db.users[key] = user;
  db.admin.stats.total_users++;
  writeDB(db);
  return res.json({ success:true, token, username:key, display_name:user.display_name, onboarding_done:false });
});

  if (!username || !password) return res.status(400).json({ error:'Missing fields' });

  const alert = selectAlert(user, `difficulty_${difficulty}`);

  const entry = {
    id: uid(), date:t,
    pages_done:        parseFloat(pages_done),
    difficulty,
    duration_minutes:  duration_minutes  || 0,
    technique_used:    technique_used    || 'standard',
    ai_alert_shown:    alert?.id || null,
    mood_score:        mood_score || 5,
  };

  user.sessions.push(entry);
  if (user.sessions.length > 90) user.sessions = user.sessions.slice(-90);

  user.progress.total_pages_memorized  = +(user.progress.total_pages_memorized + parseFloat(pages_done)).toFixed(2);
  user.progress.total_sessions_completed += 1;
  user.progress.current_page = Math.min(TOTAL_PAGES, Math.ceil(user.progress.total_pages_memorized));

  // Juz tracking
  const newJuz  = juzAtPage(user.progress.current_page);
  const prevJuz = user.progress.current_juz || 1;
  let reviewAssigned = null;
  if (newJuz > prevJuz && !user.progress.juz_completed.includes(prevJuz)) {
    user.progress.juz_completed.push(prevJuz);
    user.progress.current_juz = newJuz;
    const rv = assignReviewSystem(user.sessions, user.onboarding.daily_minutes, prevJuz);
    user.review_system = { active:true, type:rv.type, assigned_at:new Date().toISOString(),
                           trigger_juz:prevJuz, dedicated_days_per_week:rv.dedicated_days_per_week||null,
                           solidified_juz: user.review_system.solidified_juz || [] };
    reviewAssigned = rv;
    user.notifications.push({ id:uid(), type:'milestone', message:rv.message_ar, timestamp:new Date().toISOString(), read:false });
  } else {
    user.progress.current_juz = newJuz;
  }

  // Streak
  const last = user.progress.last_session_date;
  if (!last)                           { user.progress.current_streak_days = 1; }
  else if (daysBetween(last, t) === 1) { user.progress.current_streak_days++; }
  else if (daysBetween(last, t) > 1)   { user.progress.current_streak_days = 1; }
  user.progress.longest_streak_days  = Math.max(user.progress.longest_streak_days, user.progress.current_streak_days);
  user.progress.last_session_date    = t;
  user.progress.consecutive_absences = 0;

  // Phase advancement
  if (user.plan.type === 'quantum_auto') {
    user.plan.phase_days_elapsed = daysBetween(user.plan.phase_start_date, t);
    user.plan = advancePhase(user.plan);
  }

  // Energy & AFI
  const w        = db.admin.algorithm_weights;
  const delta    = difficulty === 'easy' ? 10 : difficulty === 'medium' ? 5 : -5;
  user.energy.score = Math.max(0, Math.min(100, (user.energy.score||75) + delta));
  user.energy.history.push(user.energy.score);
  if (user.energy.history.length > 10) user.energy.history.shift();
  const afiDelta = difficulty === 'easy' ? -(w.afi_skip_penalty||5) : (w.afi_complete_bonus||3);
  user.energy.afi = +Math.max(0, Math.min(100, (user.energy.afi||100) + afiDelta)).toFixed(1);

  // Notify friends
  for (const fn of (user.friends || [])) {
    const f = db.users[fn];
    if (f) f.notifications.push({ id:uid(), type:'friend_activity',
              message:`${user.display_name} أكمل جلسة الحفظ اليوم 🎯`,
              timestamp:new Date().toISOString(), read:false });
  }

  db.admin.stats.total_sessions_today++;
  db.admin.stats.total_pages_memorized_alltime = +(db.admin.stats.total_pages_memorized_alltime + parseFloat(pages_done)).toFixed(2);

  writeDB(db);

  return res.json({
    success:true, session:entry,
    progress: {
      total_pages_memorized:  user.progress.total_pages_memorized,
      current_streak_days:    user.progress.current_streak_days,
      longest_streak_days:    user.progress.longest_streak_days,
      juz_completed_count:    user.progress.juz_completed.length,
      current_juz:            user.progress.current_juz,
      current_surah:          surahAtPage(user.progress.current_page),
    },
    plan:            user.plan,
    post_alert:      selectAlert(user, 'session_end'),
    review_assigned: reviewAssigned,
    energy:          user.energy,
  });
});

  if (!db.users[w]) return res.status(404).json({ error:'User not found' });

  if (!db.chats[cid]) db.chats[cid] = { participants:[req.username, w], messages:[], last_message_at:null, motivation_score:70 };

  db.chats[cid].last_message_at = msg.timestamp;
  // Simple motivation score update
  const hoursSince = db.chats[cid].messages.length > 1
    ? (Date.now() - new Date(db.chats[cid].messages.at(-2)?.timestamp||0)) / 3600000 : 0;
  db.chats[cid].motivation_score = Math.max(10, Math.min(100, db.chats[cid].motivation_score + (hoursSince < 24 ? 5 : -10)));
  const recip = db.users[w];
  if (recip) recip.notifications.push({ id:uid(), type:'friend_activity',
    message:`رسالة جديدة من ${db.users[req.username]?.display_name||req.username}`,
    timestamp:new Date().toISOString(), read:false });
  writeDB(db);
  return res.json({ success:true, message:msg, motivation_score:db.chats[cid].motivation_score });
});
